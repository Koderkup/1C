&НаСервере
Функция ПолучитьКаталогАвтомобилей()
	РезультатФункции = Новый Соответствие;
	Запрос = Новый Запрос;
	Запрос.Текст = "
	|ВЫБРАТЬ
	|ПРЕДСТАВЛЕНИЕ(А.Марка) КАК Марка,
	|ПРЕДСТАВЛЕНИЕ(А.Кузов) КАК Кузов,
	|А.Цвет КАК Цвет
	|ИЗ
	|Справочник.Автомобили КАК А";
	Результат = Запрос.Выполнить();
	Выборка = Результат.Выбрать();
	
	Пока Выборка.Следующий() Цикл
		СоставнойКлюч = ВРег(Строка(Выборка.Марка)) + "|" + ВРег(Строка(Выборка.Кузов)) + "|" + ВРег(Строка(Выборка.Цвет));
		Структура = Новый Структура;
		Структура.Вставить("Марка", Выборка.Марка);
		Структура.Вставить("Кузов", Выборка.Кузов);
		Структура.Вставить("Цвет", Выборка.Цвет);
		РезультатФункции.Вставить(СоставнойКлюч, Структура);
	КонецЦикла;
	Возврат РезультатФункции;
КонецФункции

&НаСервере
Процедура ЗаписатьВСправочникНаСервере(МассивНовыхАвтомобилей) 
	Марки = СоответствиеМаркиАвтомобиля();
	Кузова = СоответствиеТипаКузова();
	КаталогАвтомобилей = ПолучитьКаталогАвтомобилей();

	Для Каждого Эл Из ДанныеExcel Цикл 
		ТекущийЦвет = Метаданные.Перечисления.Цвета.ЗначенияПеречисления.Найти(Эл.Цвет);
		СоставнойКлюч = ВРег(Строка(Эл.Марка)) + "|" + ВРег(Строка(Эл.Кузов)) + "|" + ?(
		ТекущийЦвет <> Неопределено, ВРег(Строка(Эл.Цвет)), ""); 
		
		Если КаталогАвтомобилей[СоставнойКлюч] <> Неопределено Тогда
			
			Продолжить;
			
		Иначе
			
			НовыйЭлементСправочника = Справочники.Автомобили.СоздатьЭлемент();
			ЗаполнитьЗначенияСвойств(НовыйЭлементСправочника, Эл);
            КаталогАвтомобилей[СоставнойКлюч] = Истина;
			Если Метаданные.Перечисления.Цвета.ЗначенияПеречисления.Найти(Эл.Цвет) <> Неопределено Тогда
				НовыйЭлементСправочника.Цвет = Перечисления.Цвета[Эл.Цвет];	
			КонецЕсли; 

			Если Марки[ВРег(Эл.Марка)] = Неопределено Тогда
			НоваяМарка = Справочники.МаркиАвтомобилей.СоздатьЭлемент();
			НоваяМарка.Наименование = Эл.Марка;
			НоваяМарка.Записать();
			НовыйЭлементСправочника.Марка = НоваяМарка.Ссылка;	
			Марки.Вставить(ВРег(Эл.Марка), НоваяМарка.Ссылка);
			Иначе
			НовыйЭлементСправочника.Марка = Марки[ВРег(Эл.Марка)];	
			КонецЕсли; 
			
			Если Кузова[ВРег(Эл.Кузов)] = Неопределено Тогда
			НовыйКузов = Справочники.ТипКузова.СоздатьЭлемент();
			НовыйКузов.Наименование = Эл.Кузов;
			НовыйКузов.Записать();
			НовыйЭлементСправочника.Кузов = НовыйКузов.Ссылка;
			Кузова.Вставить(ВРег(Эл.Кузов), НовыйКузов.Ссылка);	
			Иначе
			НовыйЭлементСправочника.Кузов = Кузова[ВРег(Эл.Кузов)];	
			КонецЕсли; 
		
			НовыйЭлементСправочника.Записать();
			МассивНовыхАвтомобилей.Добавить(НовыйЭлементСправочника.Ссылка);
		КонецЕсли;	
		КонецЦикла;	
КонецПроцедуры
